extends SceneTree
const MAIN := preload("res://scenes/main.tscn")
const SAVE := preload("res://scripts/save_manager.gd")
const BULLET := preload("res://scenes/enemy_bullet.tscn")
const BEFORE := preload("/tmp/mygame_enemy_bullet_context_before.gd")
const CONTEXT_BEFORE := preload("/tmp/mygame_enemy_projectile_frame_context_before.gd")
const CONTEXT := preload("res://scripts/enemies/enemy_projectile_frame_context.gd")
const RENDERER := preload("res://scripts/enemies/boss_projectile_renderer.gd")

func _init() -> void:
	call_deferred("_run")

func _run() -> void:
	SAVE.set_active_endless_slot(3)
	SAVE.create_or_load_endless_profile(3)
	var main = MAIN.instantiate()
	root.add_child(main)
	current_scene = main
	main.process_mode = Node.PROCESS_MODE_DISABLED
	var renderer := RENDERER.get_or_create(main)
	var reports: Array = []
	for mode in ["straight", "danmaku", "danmaku_bloom"]:
		for before in [true, false]:
			var bullets: Array = []
			for index in range(2000):
				var bullet = BULLET.instantiate()
				if before:
					bullet.set_script(BEFORE)
				main.add_child(bullet)
				bullet.reset_projectile({"position":Vector2(500, float(index % 100) - 50),"direction":Vector2.RIGHT,"speed":120.0,"damage":1.0,"lifetime":100.0,"hit_radius":8.0,"target":main.player,"motion_mode":mode,"visual_style":"boss_danmaku_violet_orb"})
				renderer.add_projectile(bullet)
				bullets.append(bullet)
			pass
			var started := Time.get_ticks_usec()
			for frame in range(50):
				var context = CONTEXT_BEFORE.new() if before else CONTEXT.new()
				for bullet in bullets:
					bullet.batch_physics_process(1.0 / 60.0, context)
			var elapsed_ms := float(Time.get_ticks_usec() - started) / 1000.0
			reports.append({"mode":mode,"before":before,"count":2000,"steps":50,"tick_total_ms":elapsed_ms,"per_step_ms":elapsed_ms / 50.0})
			for bullet in bullets:
				bullet.free()
			renderer.sync()
	print("ENEMY_BULLET_CONTEXT_AB ", JSON.stringify(reports))
	main.suppress_exit_save = true
	main.free()
	quit(0)
